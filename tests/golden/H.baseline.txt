function H(x,z){
  const d=Math.hypot(x,z);
  let h=Math.max(-12,(1-(d/(85*K))**2)*3.5-.6)+Math.sin(x*.11/K)*Math.cos(z*.13/K)*.8;
  h+=40*Math.exp(-(x*x+(z+55*K)**2)/4050);
  const t=1-sstep(30,55,Math.hypot(x-TOWN.x,z-TOWN.z));
  h=h*(1-t)+1.6*t;
  for(const I of ISL){const dx=x-I.x,dz=z-I.z,dd=Math.hypot(dx,dz);if(dd>I.R*1.6)continue;
    const v=Math.max(-12,(1-(dd/I.R)**2)*4-.6)+Math.sin(x*.09)*Math.cos(z*.11)*.7+I.pk*Math.exp(-dd*dd/(2*(I.R*.35)**2));
    if(v>h)h=v}
  {const t=1-sstep(48,68,Math.hypot(x-OUT.x,z-OUT.z));if(t>0)h=h*(1-t)+OUT.y*t} // Outlaw Town plateau
  {const ex=Math.max(0,Math.abs(x-PITCH.x)-PITCH.hw),ez=Math.max(0,Math.abs(z-PITCH.z)-PITCH.hh),t=1-sstep(0,9,Math.hypot(ex,ez));if(t>0)h=h*(1-t)+PITCH.y*t} // soccer pitch
  {const ox=x-OUT.x,oz=z-OUT.z;if(ox*ox+oz*oz<OUT.R*OUT.R){ // Outlaw Isle relief (src/games/OutlawArena.js builds on it): Lookout Ridge west, Mine Mesa east
    let a=Math.hypot(ox+125,oz-35);h+=11*Math.exp(-a*a/1352);
    a=Math.hypot(ox-125,oz-15);h+=7*(1-sstep(24,36,a))}}
  return h;
}
